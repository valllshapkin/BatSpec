class RawGraph(LocalesTrigger, ThemesTrigger, TranslateComponent, QGroupBox):

    def __init__(self, parent: QWidget | None = None):
        QGroupBox.__init__(self, parent)

        self.v = QVBoxLayout(self)
        self.v.setContentsMargins(4, 4, 4, 4)

        # горизонтальный layout внутри v → vh
        self.vh = QHBoxLayout()
        self.vh.setContentsMargins(0, 0, 0, 4)

        self.vhL = QLabel(self.tr("Display:"))
        self.vh.addWidget(self.vhL)

        self.vhC = QComboBox()
        self.vhC.addItem(self.tr("Raw (original)"), "raw")
        self.vhC.addItem(self.tr("Centered (median = 0)"), "centered")
        self.vhC.setCurrentIndex(1)
        self.vhC.currentIndexChanged.connect(self.updatePlot)
        self.vh.addWidget(self.vhC)

        self.vh.addStretch()
        self.v.addLayout(self.vh)

        self.vP = ThemedPlotWidget()
        self.vP.showGrid(x=True, y=True, alpha=0.3)
        self.v.addWidget(self.vP)

        self.vPC = self.vP.plot()
        self.vPC.setDownsampling(ds=True, auto=True, method='peak')
        self.vPC.setClipToView(True)

        RECORD_STATE.signals.recordChanged.connect(self.onRecordChanged)

    def onRecordChanged(self):
        if not RECORD_STATE.isAllExist():
            self.vPC.setData([], [])
            self.vP.setTitle(self.tr("No data"))
            return

        self.updatePlot()

    def updatePlot(self):
        if not RECORD_STATE.isAllExist():
            return

        mode = self.vhC.currentData()

        if mode == "raw":
            r = RECORD_STATE.record_raw
            s = self.tr("(original)")
        else:
            r = RECORD_STATE.record_centered
            s = self.tr("(DC offset removed)")

        if not r:
            self.vPC.setData([], [])
            return

        data = r.array
        fs = r.samplerate

        t = np.arange(len(data), dtype=np.float64) / fs

        self.vPC.setData(x=t, y=data)

        duration = t[-1] if len(t) > 0 else 0
        self.vP.setXRange(0, duration, padding=0.01)

        ymin = float(data.min())
        ymax = float(data.max())
        yrange = ymax - ymin
        pad = max(0.08, yrange * 0.12)
        self.vP.setYRange(ymin - pad, ymax + pad)

        self.vP.setTitle(self.tr("Oscillogram") + " " + s)
        self.vP.update()

    def onLanguageChange(self):
        self.vhL.setText(self.tr("Display:"))

        self.vhC.setItemText(0, self.tr("Raw (original)"))
        self.vhC.setItemText(1, self.tr("Centered (median = 0)"))

        pi = self.vP.getPlotItem()
        pi.setLabel('left',  self.tr("Amplitude"), units='FS')
        pi.setLabel('bottom', self.tr("Time"),    units='s')

        self.updatePlot()

    def onThemeChange(self):
        self.vPC.setPen(pg.mkPen(get_theme().primary, width=1))

from pathlib import Path
from PySide6.QtWidgets import (QWidget, QVBoxLayout, QTabWidget, QStackedWidget,
                               QLabel, QListWidget, QListWidgetItem, QPushButton)
from PySide6.QtCore import Slot, Qt
from PySide6.QtGui import QIcon

from MySide6.TabInst import TabInstance

from BatSpec.Services import TranslateComponent

from ..Logic import RECORD_STATE
from BatSpec.Modules.Project.Logic import PROJECT_STATE


class Load(LocalesTrigger, ThemesTrigger, TranslateComponent, TabInstance):
    def __init__(self, parent: QWidget | None = None, tab_widget: QTabWidget | None = None) -> None:
        TabInstance.__init__(self, parent, tab_widget)
        
        self.v = QVBoxLayout(self)
        
        # QStackedWidget для переключения между "Заглушкой" и "Списком"
        self.vS = QStackedWidget()
        self.v.addWidget(self.vS)
        
        # --- СТРАНИЦА 0: Заглушка ---
        self.lbl_no_project = QLabel()
        self.lbl_no_project.setAlignment(Qt.AlignmentFlag.AlignCenter)
        self.vS.addWidget(self.lbl_no_project)
        
        # --- СТРАНИЦА 1: Рабочая область со списком ---
        self.page_list = QWidget()
        self.page_list_layout = QVBoxLayout(self.page_list)
        self.page_list_layout.setContentsMargins(0, 0, 0, 0)
        
        self.list_records = QListWidget()
        self.list_records.itemDoubleClicked.connect(self.on_item_selected)
        self.page_list_layout.addWidget(self.list_records)
        
        self.btn_refresh = QPushButton()
        self.btn_refresh.clicked.connect(self.refresh_list)
        self.page_list_layout.addWidget(self.btn_refresh)
        
        self.vS.addWidget(self.page_list)
        
        # --- ПОДПИСКИ НА СИГНАЛЫ ---
        PROJECT_STATE.projectNewLoaded.connect(self.update_ui_state)
        
        # Инициализация
        self.update_ui_state()

    @Slot()
    def update_ui_state(self):
        """Переключает экраны в зависимости от того, загружен ли проект"""
        if PROJECT_STATE.currentProject is None:
            self.vS.setCurrentIndex(0)
            self.list_records.clear()
        else:
            self.vS.setCurrentIndex(1)
            self.refresh_list()

    @Slot()
    def refresh_list(self):
        """Обновляет список записей из папки проекта"""
        self.list_records.clear()
        
        proj = PROJECT_STATE.currentProject
        if not proj:
            return
            
        records_dir = proj.folder_records
        
        # Защита: если папки еще нет, просто выходим
        if not records_dir.exists():
            return
            
        # Сканируем папку. Ищем структуру: folder_records / {name} / record.wav
        for item_dir in records_dir.iterdir():
            if item_dir.is_dir():
                wav_path = item_dir / "record.wav"
                
                if wav_path.exists() and wav_path.is_file():
                    # Создаем элемент списка. Имя - это название папки {name}
                    item = QListWidgetItem(item_dir.name)
                    item.setIcon(QIcon.fromTheme("audio-x-generic"))
                    
                    # Сохраняем полный путь к WAV файлу в data (роль 256, как в Import)
                    item.setData(256, str(wav_path.absolute()))
                    
                    self.list_records.addItem(item)

from numpy.typing import NDArray
import numpy as np
import soundfile as sf
from PySide6.QtCore import QObject, Signal
from pathlib import Path
import logging

# Настройка логгера для модуля
logger = logging.getLogger(__name__)


class Record:
    def __init__(self, array: NDArray[np.float64], samplerate: int, path: Path | None = None, 
                 original_channels: int = 1, original_subtype: str = 'UNKNOWN'):
        self.array = array
        self.samplerate = samplerate
        self.path = path
        self.original_channels = original_channels
        self.original_subtype = original_subtype

    @property
    def duration(self) -> float:
        """Длительность в секундах"""
        return len(self.array) / self.samplerate


class _RecordState:

    class Signals(QObject):
        recordChanged = Signal()
    
    record_raw: Record | None = None
    record_centered: Record | None = None

    def __init__(self):
        self.signals = self.Signals()

    def isAllExist(self) -> bool:
        if self.record_raw and self.record_centered:
            return True
        return False


    def import_record(self, path: Path):
        try:
            if not path.exists():
                raise FileNotFoundError(f"Файл не найден: {path}")

            info = sf.info(str(path))
            subtype = info.subtype

            data, samplerate = sf.read(str(path), dtype='float64', always_2d=False)

            original_channels = 1 if data.ndim == 1 else data.shape[1]

            # В моно
            if data.ndim == 2 and data.shape[1] > 1:
                data = np.mean(data, axis=1)
            elif data.ndim == 2 and data.shape[1] == 1:
                data = data.flatten()

            # ────────────────────────────────────────────────
            # Основная запись (как была)
            self.record_raw = Record(
                array=data.copy(),   # важно сделать копию!
                samplerate=samplerate,
                path=path,
                original_channels=original_channels,
                original_subtype=subtype
            )

            # Вторая запись — с удалённым смещением (медиана ≈ 0)
            median_offset = np.median(data)
            data_zero = data - median_offset

            self.record_centered = Record(
                array=data_zero,
                samplerate=samplerate,
                path=path,                      # можно оставить тот же путь
                original_channels=original_channels,
                original_subtype=subtype
            )
            # ────────────────────────────────────────────────

            logger.info(f"Загружен файл: {path.name}")
            logger.info(f"  Subtype: {subtype}")
            logger.info(f"  Частота: {samplerate} Гц")
            logger.info(f"  Каналов (исх.): {original_channels}")
            logger.info(f"  Длительность: {self.record_raw.duration:.2f} с")

            logger.info(f"  Оригинал    min / max / median : {data.min():.5f} / {data.max():.5f} / {np.median(data):.5f}")
            logger.info(f"  Нулевой DC  min / max / median : {data_zero.min():.5f} / {data_zero.max():.5f} / {np.median(data_zero):.5f}")

            if not self.isAllExist(): raise RuntimeError()

        except Exception as e:
            logger.exception("Ошибка импорта записи")
            self.record_raw = None
            self.record_centered = None
        finally:
            self.signals.recordChanged.emit()

# Глобальный экземпляр состояния
RECORD_STATE = _RecordState()

class PathChecker(QThread):
    """Поток для фоновой проверки актуальности путей"""
    
    paths_checked = Signal(dict)  # {path_str: exists}
    
    def __init__(self, paths: List[str]):

            return raw
        if isinstance(raw, str) and raw:

        if not ev.isExit():
            pos = self.mapSceneToView(ev.scenePos())
            self.mouseMoved.emit(pos.x(), pos.y())

    def mouseDragEvent(self, ev, axis=None):
        if self.roi_mode != 'move' and ev.button() == Qt.MouseButton.LeftButton:
            ev.accept()
            pos = self.mapToView(ev.pos())
            sp  = self.mapToView(ev.buttonDownPos())
            if ev.isStart():
                if self.roi_mode == 'rect':
                    self.current_roi = pg.RectROI(sp, [0,0], pen='y', removable=True)
                    self.current_roi.addScaleHandle([1,1],[0,0])
                elif self.roi_mode == 'line':
                    self.current_roi = pg.LineSegmentROI([sp, pos], pen='y', removable=True)
                self.current_roi.sigRemoveRequested.connect(lambda r: self.removeItem(r))
                self.addItem(self.current_roi)
            elif ev.isFinish():
                self.current_roi = None
            else:
                if self.current_roi:
                    if self.roi_mode == 'rect':
                        x0,y0,x1,y1 = sp.x(),sp.y(),pos.x(),pos.y()
                        self.current_roi.setPos([min(x0,x1), min(y0,y1)])
                        self.current_roi.setSize([abs(x1-x0), abs(y1-y0)])
                    elif self.roi_mode == 'line':
                        self.current_roi.movePoint(self.current_roi.getHandles()[1], pos)
            return
        super().mouseDragEvent(ev, axis)

from typing import Tuple
import tensorflow as tf
import numpy as np
from PySide6.QtCore import QObject, Signal, QRunnable, QThreadPool

from ..Record.Logic import RECORD_STATE

def make_spec(
    signal: np.ndarray, 
    window_array: np.ndarray, 
    frame_step: int, 
    fft_length: int,      
    sample_rate: int,
    device: str
) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:

    frame_length = len(window_array)

    def win_func(length, dtype=tf.float32):
        assert length == frame_length
        return tf.convert_to_tensor(window_array, dtype=dtype)

    with tf.device(device):
        stft = tf.signal.stft(
            signal,
            frame_length=frame_length,
            frame_step=frame_step,
            fft_length=fft_length,    
            window_fn=win_func 
        )
        spectrogram = np.abs(stft.numpy())

    freq_axis = np.fft.rfftfreq(fft_length, d=1.0/sample_rate)
    num_frames = stft.shape[0]
    time_axis = (np.arange(num_frames) * frame_step) / sample_rate

    # Считаем RMS (Корень из интеграла мощности по частоте для каждого временного фрейма)
    rms = np.sqrt(np.sum(spectrogram**2, axis=1))

    return spectrogram, time_axis, freq_axis, rms


# --- Воркер теперь передает 4 параметра (добавлен массив RMS) ---
class STFTWorkerSignals(QObject):
    finished = Signal(object, float, float, object) # db_matrix, duration, max_freq, rms_array
    error = Signal(str)

class STFTDataWorker(QRunnable):
    def __init__(self, state_instance):
        super().__init__()
        self.state = state_instance
        self.signals = STFTWorkerSignals()

    def run(self):
        try:
            spectrogram, time_axis, freq_axis, rms = self.state.makeSpec()
            db = 20.0 * np.log10(spectrogram + 1e-10)
            db_t = np.ascontiguousarray(db.T, dtype=np.float32)
            duration = float(time_axis[-1]) if len(time_axis) > 0 else 0.0
            max_freq = float(freq_axis[-1]) if len(freq_axis) > 0 else 0.0
            
            self.signals.finished.emit(db_t, duration, max_freq, rms)
        except Exception as e:
            self.signals.error.emit(str(e))

import matplotlib.pyplot as plt
import scipy.signal

from typing import Tuple
import tensorflow as tf
import numpy as np

def make_spec(
    signal: np.ndarray, 
    window_array: np.ndarray, 
    frame_step: int, 
    sample_rate: int  # Добавлен параметр для расчета времени и частоты
) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:

    frame_length = len(window_array)

    def win_func(length, dtype=tf.float32):
        assert length == frame_length
        return tf.convert_to_tensor(window_array, dtype=dtype)

    # 1. Вычисляем STFT
    stft = tf.signal.stft(
        signal,
        frame_length=frame_length,
        frame_step=frame_step,
        window_fn=win_func 
    )

    # 2. Конвертация в numpy и получение амплитудной спектрограммы (Magnitude)
    # stft возвращает комплексные числа. Берем модуль (abs), чтобы получить амплитуду.
    # Транспонируем (.T), чтобы получить форму [частоты, время] — это стандарт для отрисовки.
    spectrogram = np.abs(stft.numpy()).T 

    # 3. Генерация оси частот (Y-axis) в Герцах
    # По умолчанию TF использует fft_length равный ближайшей степени 2 >= frame_length
    fft_length = 2 ** int(np.ceil(np.log2(frame_length)))
    
    # np.fft.rfftfreq идеально подходит, так как генерирует частоты для 
    # уникальных бинов (от 0 до Найквиста), что совпадает с выводом TF
    freq_axis = np.fft.rfftfreq(fft_length, d=1.0/sample_rate)

    # 4. Генерация оси времени (X-axis) в Секундах
    # stft.shape[0] - количество получившихся фреймов (окон)
    num_frames = stft.shape[0]
    
    # Время начала каждого фрейма
    time_axis = (np.arange(num_frames) * frame_step) / sample_rate

    return spectrogram, time_axis, freq_axis

from __future__ import annotations

import numpy as np
from dataclasses import dataclass, field
from scipy.ndimage import (gaussian_filter1d, label, binary_closing,
                           laplace, gaussian_filter, binary_opening,
                           binary_dilation, binary_erosion)
from PySide6.QtCore import QObject, Signal, QRunnable, QThreadPool
import logging

from ..STFTModel.Logic import STFT_MODEL_STATE

logger = logging.getLogger(__name__)


# ---------------------------------------------------------------------------
# Data structures
# ---------------------------------------------------------------------------

@dataclass
class CallTrace:
    ctrl_t: np.ndarray
    ctrl_f: np.ndarray
    t0: float = 0.0
    t1: float = 0.0
    f0: float = 0.0
    f1: float = 0.0


@dataclass
class ProcessingResult:
    original:      np.ndarray
    normalized:    np.ndarray
    time_power:    np.ndarray
    duration:      float = 0.0
    max_freq:      float = 0.0
    calls:         list[CallTrace] = field(default_factory=list)
    rms:           np.ndarray = field(default_factory=lambda: np.array([]))
    laplacian:     np.ndarray = field(default_factory=lambda: np.array([]))
    binary:        np.ndarray = field(default_factory=lambda: np.array([]))
    after_erosion: np.ndarray = field(default_factory=lambda: np.array([]))
    after_dilation:np.ndarray = field(default_factory=lambda: np.array([]))


# ---------------------------------------------------------------------------
# Normalisation
# ---------------------------------------------------------------------------

def normalize_globally(db_matrix: np.ndarray, rms: np.ndarray,
                       percentile: float) -> np.ndarray:
    noise_threshold = np.percentile(rms, percentile)
    noise_idx = np.where(rms <= noise_threshold)[0]
    if len(noise_idx) < 2:
        logger.warning("Too few noise frames — skipping global normalisation.")
        return db_matrix.copy()
    noise_frames = db_matrix[:, noise_idx]
    mean = np.mean(noise_frames, axis=1)
    std  = np.std(noise_frames, axis=1)
    std[std < 1e-6] = 1.0
    return (db_matrix - mean[:, np.newaxis]) / std[:, np.newaxis]


# ---------------------------------------------------------------------------
# Laplacian
# ---------------------------------------------------------------------------

def compute_laplacian(norm: np.ndarray, pre_smooth: float = 1.0) -> np.ndarray:
    if pre_smooth > 0:
        smoothed = gaussian_filter(norm.astype(np.float64), sigma=pre_smooth)
    else:
        smoothed = norm.astype(np.float64)
    lap = laplace(smoothed).astype(np.float32)
    np.nan_to_num(lap, copy=False)
    return lap


# ---------------------------------------------------------------------------
# Morphological pipeline  (3 binary maps)
# ---------------------------------------------------------------------------

def _disk_struct(radius: int) -> np.ndarray:
    r = max(1, radius)
    y, x = np.ogrid[-r:r+1, -r:r+1]
    return (x**2 + y**2 <= r**2)


def morpho_pipeline(
    lap:            np.ndarray,
    threshold_std:  float = 1.0,
    erosion_radius: int   = 2,
    dilation_radius:int   = 3,
) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Три бинарные карты:
    1. binary        — после порога: |lap| > threshold_std * std
    2. after_erosion — после эрозии: убиваем мелкий лес
    3. after_dilation— после нарощения: восстанавливаем форму выбоин

    Возвращает три uint8 массива (0/1).
    """
    src = np.abs(lap).astype(np.float32)

    # 1. Бинаризация
    thr    = threshold_std * src.std()
    binary = (src > thr).astype(np.uint8)

    # 2. Эрозия — убиваем всё мелкое (лес)
    struct_e  = _disk_struct(erosion_radius)
    eroded    = binary_erosion(binary, structure=struct_e).astype(np.uint8)

    # 3. Нарощение — возвращаем форму тому что выжило
    struct_d  = _disk_struct(dilation_radius)
    dilated   = binary_dilation(eroded, structure=struct_d).astype(np.uint8)

    return binary, eroded, dilated


# ---------------------------------------------------------------------------
# Ridge detection
# ---------------------------------------------------------------------------

def extract_calls(
    normalized:       np.ndarray,
    duration:         float,
    max_freq:         float,
    time_thresh:      float,
    gauss_sigma_ms:   float,
    gauss_window_ms:  float,
    merge_gap_ms:     float,
    min_dur_ms:       float,
) -> tuple[np.ndarray, list[CallTrace]]:
    F, T = normalized.shape
    sec_per_bin = duration / T if T > 0 else 1.0
    hz_per_bin  = max_freq  / F if F > 0 else 1.0

    raw_power  = np.mean(normalized ** 2, axis=0)
    sigma_bins = max((gauss_sigma_ms  / 1000.0) / sec_per_bin, 0.5)
    trunc      = max((gauss_window_ms / 1000.0) / sec_per_bin / 2.0 / sigma_bins, 1.0)
    smoothed   = gaussian_filter1d(raw_power, sigma=sigma_bins, truncate=trunc)

    mask         = smoothed > time_thresh
    gap_bins     = max(1, int(round((merge_gap_ms / 1000.0) / sec_per_bin)))
    min_dur_bins = max(1, int(round((min_dur_ms   / 1000.0) / sec_per_bin)))
    mask         = binary_closing(mask, structure=np.ones(gap_bins, dtype=bool))
    labeled, n_segs = label(mask)

    calls: list[CallTrace] = []
    for seg_id in range(1, n_segs + 1):
        cols = np.where(labeled == seg_id)[0]
        if len(cols) < min_dur_bins:
            continue
        ridge_f_bins = np.argmax(normalized[:, cols], axis=0)
        t0 = float(cols[0]  * sec_per_bin)
        t1 = float(cols[-1] * sec_per_bin)
        f0 = float(ridge_f_bins.min() * hz_per_bin)
        f1 = float(ridge_f_bins.max() * hz_per_bin)
        calls.append(CallTrace(
            ctrl_t=cols * sec_per_bin,
            ctrl_f=ridge_f_bins * hz_per_bin,
            t0=t0, t1=t1, f0=f0, f1=f1,
        ))

    return smoothed, calls


# ---------------------------------------------------------------------------
# Worker
# ---------------------------------------------------------------------------

class _DetectSignals(QObject):
    finished = Signal(object)
    error    = Signal(str)


class DetectWorker(QRunnable):
    def __init__(self, state: _ProcessingState):
        super().__init__()
        self.state   = state
        self.signals = _DetectSignals(parent=state)

    def run(self):
        try:
            raw_db   = self.state.raw_db
            original = np.ascontiguousarray(raw_db, dtype=np.float32)
            np.nan_to_num(original, copy=False, nan=0.0, posinf=0.0, neginf=0.0)

            # 1. Нормализация на шум
            norm = normalize_globally(original, self.state.raw_rms,
                                      self.state.percentile)
            norm = np.ascontiguousarray(norm, dtype=np.float32)
            np.nan_to_num(norm, copy=False, nan=0.0, posinf=0.0, neginf=0.0)

            # 2. Лапласиан
            lap = compute_laplacian(norm, pre_smooth=self.state.lap_pre_smooth)

            # 3. Морфологический пайплайн — три бинарные карты
            binary, after_erosion, after_dilation = morpho_pipeline(
                lap,
                threshold_std   = self.state.morpho_threshold_std,
                erosion_radius  = self.state.morpho_erosion_radius,
                dilation_radius = self.state.morpho_dilation_radius,
            )

            # 4. Ridge detection на нормализованной спектрограмме
            smoothed, calls = extract_calls(
                normalized      = norm,
                duration        = self.state.raw_duration,
                max_freq        = self.state.raw_max_freq,
                time_thresh     = self.state.time_thresh,
                gauss_sigma_ms  = self.state.gauss_sigma_ms,
                gauss_window_ms = self.state.gauss_window_ms,
                merge_gap_ms    = self.state.merge_gap_ms,
                min_dur_ms      = self.state.min_dur_ms,
            )

            self.signals.finished.emit(ProcessingResult(
                original       = original,
                normalized     = norm,
                time_power     = smoothed,
                duration       = self.state.raw_duration,
                max_freq       = self.state.raw_max_freq,
                calls          = calls,
                rms            = self.state.raw_rms,
                laplacian      = lap,
                binary         = binary,
                after_erosion  = after_erosion,
                after_dilation = after_dilation,
            ))
        except Exception:
            logger.exception("DetectWorker error")
            import traceback
            self.signals.error.emit(traceback.format_exc())


# ---------------------------------------------------------------------------
# State
# ---------------------------------------------------------------------------

class _ProcessingState(QObject):
    processingStarted  = Signal()
    processingFinished = Signal(object)
    processingError    = Signal(str)
    dirtyStateChanged  = Signal(bool)

    def __init__(self):
        super().__init__()

        self.percentile:      float = 5.0
        self.lap_pre_smooth:  float = 1.0

        self.morpho_threshold_std:   float = 1.0
        self.morpho_erosion_radius:  int   = 2
        self.morpho_dilation_radius: int   = 3

        self.time_thresh:     float = 3.0
        self.gauss_sigma_ms:  float = 4.0
        self.gauss_window_ms: float = 8.0
        self.merge_gap_ms:    float = 2.0
        self.min_dur_ms:      float = 2.0

        self.raw_db:       np.ndarray | None = None
        self.raw_rms:      np.ndarray | None = None
        self.raw_duration: float             = 0.0
        self.raw_max_freq: float             = 0.0

        self.result:   ProcessingResult | None = None
        self.is_dirty: bool = True

        self.threadpool = QThreadPool()
        STFT_MODEL_STATE.calculationFinished.connect(self._on_stft_finished)

    def set_detect_parameters(self, *, percentile: float, time_thresh: float,
                              gauss_sigma_ms: float, gauss_window_ms: float,
                              merge_gap_ms: float, min_dur_ms: float) -> None:
        self.percentile      = percentile
        self.time_thresh     = max(1.0, time_thresh)
        self.gauss_sigma_ms  = max(0.1, gauss_sigma_ms)
        self.gauss_window_ms = max(gauss_sigma_ms, gauss_window_ms)
        self.merge_gap_ms    = max(0.0, merge_gap_ms)
        self.min_dur_ms      = max(0.1, min_dur_ms)
        self.is_dirty = True
        self.dirtyStateChanged.emit(True)

    def set_laplacian_parameters(self, *, pre_smooth: float) -> None:
        self.lap_pre_smooth = max(0.0, pre_smooth)
        self.is_dirty = True
        self.dirtyStateChanged.emit(True)

    def set_morpho_parameters(self, *, threshold_std: float,
                              erosion_radius: int,
                              dilation_radius: int) -> None:
        self.morpho_threshold_std   = max(0.1, threshold_std)
        self.morpho_erosion_radius  = max(1, erosion_radius)
        self.morpho_dilation_radius = max(1, dilation_radius)
        self.is_dirty = True
        self.dirtyStateChanged.emit(True)

    def apply_detect(self) -> None:
        if self.raw_db is not None:
            self._run_detect()

    def _on_stft_finished(self, db_t, duration, max_freq, rms):
        self.raw_db       = db_t
        self.raw_duration = duration
        self.raw_max_freq = max_freq
        self.raw_rms      = rms
        self.is_dirty = False
        self.dirtyStateChanged.emit(False)
        self._run_detect()

    def _run_detect(self) -> None:
        self.processingStarted.emit()
        worker = DetectWorker(self)
        worker.signals.finished.connect(self._on_detect_done)
        worker.signals.error.connect(self.processingError.emit)
        self.threadpool.start(worker)

    def _on_detect_done(self, result: ProcessingResult) -> None:
        self.result = result
        self.is_dirty = False
        self.dirtyStateChanged.emit(False)
        self.processingFinished.emit(result)


PROCESSING_STATE = _ProcessingState()
